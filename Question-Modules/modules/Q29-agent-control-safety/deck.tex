\begin{frame}{Question}\begin{block}{Q29}Is AI controllable, and how do agentic workflows and safety protocols constrain it?\end{block}\end{frame}
\begin{frame}{Control is a systems property}\begin{itemize}\item Model + prompts + tools + credentials + workflow + people\item State a bounded control claim, not ``the model is safe.''\item Ask: authority, trust boundaries, stop/recovery paths.\end{itemize}\end{frame}
\begin{frame}{Least privilege}\begin{itemize}\item Read vs write; staging vs production\item Scoped, expiring credentials\item High-impact action bound to verified approval\end{itemize}\alert{A tool interface is not a trust guarantee.}\end{frame}
\begin{frame}{Prompts are not access control}\begin{itemize}\item Retrieved content can carry indirect prompt injection.\item Treat external text as data, not authority.\item Enforce policy and arguments with deterministic code.\end{itemize}\end{frame}
\begin{frame}{Defense in depth}\begin{itemize}\item Capability/egress limits and sandboxing\item Argument validation and approval binding\item Logging, rate limits, and reversible actions\item Human confirmation for consequential side effects\end{itemize}\end{frame}
\begin{frame}{Evaluate the workflow}\begin{itemize}\item Normal, ambiguous, unauthorized, malformed, injected cases\item Success, false approval/block, intervention, cost, severity\item Preserve held-out tests and environment versions\end{itemize}\end{frame}
\begin{frame}{Human oversight}\begin{itemize}\item Show target, diff, evidence, and consequences.\item Avoid rubber-stamp queues and automation bias.\item Make escalation and stop legitimate outcomes.\end{itemize}\end{frame}
\begin{frame}{Takeaway}\begin{block}{Calibrated delegation}Minimize authority, validate side effects, test boundaries, and reserve stronger controls for irreversible or high-impact actions.\end{block}\end{frame}\end{document}
