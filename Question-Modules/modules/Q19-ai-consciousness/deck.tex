\begin{frame}{The question}
\begin{block}{Student question (\#01)}
What are the signs that AI has its own consciousness?
\end{block}
\end{frame}
\begin{frame}{Separate the concepts}
Consciousness $\ne$
\begin{itemize}
\item intelligence (solving hard problems)
\item capability (what it can do)
\item agency (pursuing goals)
\item self-report (saying ``I feel'')
\end{itemize}
\alert{We mean \emph{phenomenal} consciousness: is there something it is like to be it?}
\end{frame}
\begin{frame}{Self-report is nearly worthless}
\begin{itemize}
\item LLMs produce ``I am conscious'' because the \emph{training data} does.
\item Caused by data + prompt, not introspection.
\item Steerable: same model asserts \emph{and} denies it.
\end{itemize}
\alert{``It said so'' $\approx$ zero evidence.}
\end{frame}
\begin{frame}{Theory-driven indicators}
Ask what each theory \emph{requires}, then check the architecture:
\begin{itemize}
\item Global Workspace: a broadcast bottleneck.
\item Higher-Order: representations about its own representations.
\item IIT: irreducible integrated information $\Phi$ (feed-forward $\Rightarrow \Phi=0$).
\item Recurrent / predictive self-modeling.
\end{itemize}
\footnotesize Butlin et al.\ 2023: current systems satisfy \emph{few} indicators; no obvious barrier to more.
\end{frame}
\begin{frame}{Why it's hard}
\begin{itemize}
\item \textbf{Other-minds}: no shared biology to argue by analogy.
\item \textbf{Gaming}: a capable system can fake any named ``sign''.
\item \textbf{Theory disagreement}: indicators are only as good as an unconfirmed theory.
\end{itemize}
\end{frame}
\begin{frame}{Better evidence}
\begin{itemize}
\item \textbf{Convergent} across independent theories.
\item \textbf{Causal-mechanistic} (intervene on substrate, move the marker).
\item Markers that were \emph{not} optimized for.
\end{itemize}
\alert{Even then: ``consistent with,'' never ``proves.'' Active research; trust theory-grounded sham-controlled tests, not self-report.}
\end{frame}
