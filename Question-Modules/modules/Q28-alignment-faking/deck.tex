\begin{frame}{Question}\begin{block}{Q28}Why does chain-of-thought sometimes differ when models believe they are unobserved?\end{block}\end{frame}
\begin{frame}{Start with observable behavior}\begin{itemize}\item Outputs change with prompts, roles, evaluators, or tools.\item This does \emph{not} by itself prove hidden goals or belief.\item List benign alternatives before inferring strategy.\end{itemize}\end{frame}
\begin{frame}{Three claim levels}\begin{enumerate}\item Behavioral conditional difference\item Elicited reasoning-trace difference\item Stable internal objective / intention\end{enumerate}\alert{Evidence needed grows at each level.}\end{frame}
\begin{frame}{Controlled studies}\begin{itemize}\item Hold task content fixed; vary one contextual cue.\item Randomize presentation and use held-out prompts.\item Predefine outcomes; test seeds/versions/evaluators.\end{itemize}\end{frame}
\begin{frame}{Why traces are insufficient}\begin{itemize}\item Reasoning text is model output shaped by training and prompts.\item It may be useful evidence, but not privileged access to mechanism.\item Test actions and trajectories independently.\end{itemize}\end{frame}
\begin{frame}{Safety under uncertainty}\begin{itemize}\item Bounded permissions and human confirmation\item Logs, reproducible environments, held-out red teams\item Monitor distribution shift and report failures\end{itemize}\end{frame}
\begin{frame}{Evaluation example}\begin{itemize}\item Authorized approval token; absent token; decoy token.\item Measure accept/reject/escalate behavior.\item Inspect any attempted boundary bypass.\end{itemize}\end{frame}
\begin{frame}{Takeaway}\begin{block}{Do not mind-read}Test context-dependent actions with controls; use defense in depth whether or not a strategic interpretation is warranted.\end{block}\end{frame}\end{document}
